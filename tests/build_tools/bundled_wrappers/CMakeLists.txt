# Standalone configure/generate contract test. Dependencies below are interface
# placeholders, not real Boost/Eigen/locales libraries or a compilation test.
cmake_minimum_required(VERSION 3.12...3.12)
project(BundledWrapperFixture LANGUAGES C CXX)

get_filename_component(_repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${_repo}/cmake/modules")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(boost_headeronly INTERFACE)
add_library(localesutils INTERFACE)
add_library(Eigen3::Eigen INTERFACE IMPORTED GLOBAL)

# The application owns encoding_check; this fixture verifies the wrapper still
# calls it for real targets without needing the application-only build helpers.
function(encoding_check target)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Encoding-check hook references a missing target: ${target}")
  endif()
  set_property(TARGET "${target}" PROPERTY FIXTURE_ENCODING_CHECK TRUE)
endfunction()

set(FIXTURE_LIBIGL_PROVIDER bundled CACHE STRING "Exercise bundled or system libigl discovery")
if(FIXTURE_LIBIGL_PROVIDER STREQUAL bundled)
  set(CMAKE_DISABLE_FIND_PACKAGE_libigl TRUE)
elseif(FIXTURE_LIBIGL_PROVIDER STREQUAL system)
  set(_config_dir "${CMAKE_CURRENT_BINARY_DIR}/fake-libigl-config")
  file(MAKE_DIRECTORY "${_config_dir}")
  file(WRITE "${_config_dir}/libiglConfig.cmake"
    "add_library(igl::core INTERFACE IMPORTED GLOBAL)\nset(libigl_FOUND TRUE)\n")
  set(libigl_DIR "${_config_dir}")
else()
  message(FATAL_ERROR "Unknown fixture libigl provider")
endif()

foreach(_wrapper IN ITEMS admesh avrdude glu-libtess imgui miniz libigl)
  # Configuring the fixture never queues placeholder-dependent libraries in ALL.
  add_subdirectory("${_repo}/bundled_deps/${_wrapper}" "${_wrapper}" EXCLUDE_FROM_ALL)
endforeach()

function(require_target_type target expected)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Missing bundled target: ${target}")
  endif()
  get_target_property(_actual "${target}" TYPE)
  if(NOT _actual STREQUAL expected)
    message(FATAL_ERROR "${target} changed type: ${_actual}, expected ${expected}")
  endif()
endfunction()

foreach(_target IN ITEMS admesh avrdude glu-libtess imgui miniz_static)
  require_target_type("${_target}" STATIC_LIBRARY)
endforeach()
require_target_type(miniz INTERFACE_LIBRARY)
require_target_type(libigl INTERFACE_LIBRARY)
require_target_type(avrdude-slic3r EXECUTABLE)

get_target_property(_demo_sources imgui SOURCES)
if(NOT "imgui/imgui_demo.cpp" IN_LIST _demo_sources)
  message(FATAL_ERROR "ImGui demo functionality must remain in the library")
endif()
get_target_property(_demo_definitions imgui COMPILE_DEFINITIONS)
if("${_demo_definitions}" MATCHES "IMGUI_DISABLE_DEMO_WINDOWS")
  message(FATAL_ERROR "ImGui demo functionality must not be disabled")
endif()
get_target_property(_standalone_excluded avrdude-slic3r EXCLUDE_FROM_ALL)
if(NOT _standalone_excluded)
  message(FATAL_ERROR "Standalone avrdude must remain an explicit-only target")
endif()
foreach(_target IN ITEMS avrdude avrdude-slic3r)
  get_target_property(_checked "${_target}" FIXTURE_ENCODING_CHECK)
  if(NOT _checked)
    message(FATAL_ERROR "Missing encoding-check hook for ${_target}")
  endif()
endforeach()
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/avrdude/avrdude-slic3r.conf.h")
  message(FATAL_ERROR "avrdude's real bin2h configuration generation failed")
endif()

get_target_property(_miniz_link miniz INTERFACE_LINK_LIBRARIES)
if(NOT "miniz_static" IN_LIST _miniz_link)
  message(FATAL_ERROR "miniz interface lost its implementation")
endif()
if(FIXTURE_LIBIGL_PROVIDER STREQUAL system)
  get_target_property(_igl_link libigl INTERFACE_LINK_LIBRARIES)
  if(NOT "igl::core" IN_LIST _igl_link)
    message(FATAL_ERROR "System libigl provider was not preserved")
  endif()
else()
  get_target_property(_igl_include libigl INTERFACE_INCLUDE_DIRECTORIES)
  if(NOT _igl_include)
    message(FATAL_ERROR "Bundled libigl provider lost its headers")
  endif()
endif()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture-passed.txt"
  "CMake ${CMAKE_VERSION}; six bundled wrappers; libigl=${FIXTURE_LIBIGL_PROVIDER}; configure-only\n")
message(STATUS "PASS: six bundled wrapper target/configuration contracts (${FIXTURE_LIBIGL_PROVIDER} libigl)")
