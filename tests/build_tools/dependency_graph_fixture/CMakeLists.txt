cmake_minimum_required(VERSION 3.19)
project(DependencyGraphContract NONE)
get_filename_component(_repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_SYSTEM_NAME "${GRAPH_SYSTEM}")
include("${_repo}/deps/DependencyGraph.cmake")
if(FAIL_MISSING)
  prusaslicer_require_dependency_target(Consumer Missing)
endif()
file(GLOB _recipes "${_repo}/deps/+*/*.cmake")
foreach(_recipe IN LISTS _recipes)
  file(STRINGS "${_recipe}" _duplicate REGEX "^[ \t]*(set|list)[ \t]*\\([ \t]*(APPEND[ \t]+)?DEP_[A-Za-z0-9_]+_DEPENDS")
  if(_duplicate)
    message(FATAL_ERROR "Recipe redefines authoritative dependency graph: ${_recipe}")
  endif()
endforeach()
file(GLOB _packages RELATIVE "${_repo}/deps" "${_repo}/deps/+*")
foreach(_package IN LISTS _packages)
  string(SUBSTRING "${_package}" 1 -1 _name)
  add_custom_target(dep_${_name})
endforeach()
foreach(_package IN LISTS _packages)
  string(SUBSTRING "${_package}" 1 -1 _name)
  foreach(_child IN LISTS DEP_${_name}_DEPENDS)
    prusaslicer_require_dependency_target(${_name} ${_child})
  endforeach()
endforeach()
prusaslicer_dependency_closure(_closure OpenVDB)
if(NOT "${_closure}" STREQUAL "Blosc;Boost;OpenVDB;TBB;ZLIB")
  message(FATAL_ERROR "Wrong OpenVDB closure: ${_closure}")
endif()
prusaslicer_dependency_closure(_closure CURL)
if(GRAPH_SYSTEM STREQUAL "Linux")
  set(_expected "CURL;OpenSSL;ZLIB")
else()
  set(_expected "CURL;ZLIB")
endif()
if(NOT "${_closure}" STREQUAL "${_expected}")
  message(FATAL_ERROR "Wrong CURL closure: ${_closure}")
endif()
# Explicit release-provided dependencies remain valid in nested Debug graphs.
set(SYSTEM_PROVIDED_PACKAGES ReleaseProvided)
prusaslicer_require_dependency_target(Consumer ReleaseProvided)
