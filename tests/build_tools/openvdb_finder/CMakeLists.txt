cmake_minimum_required(VERSION 3.15)
project(OpenVDBFinderFixture LANGUAGES NONE)

get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_MODULE_PATH "${REPO_ROOT}/cmake/modules")
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG TRUE)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CONFIGURATION_TYPES Release)
set(OPENVDB_USE_STATIC_LIBS ON)

# This fixture tests discovery and imported targets without invoking a compiler.
set(LLVM_COMPILER_IS_GCC_COMPATIBLE FALSE)
set(HAVE_CXX_ATOMICS64_WITHOUT_LIB TRUE CACHE INTERNAL "")
set(LLVM_HAS_ATOMICS TRUE CACHE INTERNAL "")
set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig TRUE)
function(pkg_check_modules prefix)
  set(${prefix}_FOUND FALSE PARENT_SCOPE)
endfunction()

set(_prefix "${CMAKE_CURRENT_BINARY_DIR}/prefix")
set(CMAKE_PREFIX_PATH "${_prefix}")
foreach(_package TBB ZLIB Boost IlmBase Imath)
  file(MAKE_DIRECTORY "${_prefix}/lib/cmake/${_package}")
endforeach()
file(WRITE "${_prefix}/lib/cmake/TBB/TBBConfig.cmake"
  "add_library(TBB::tbb INTERFACE IMPORTED)\nset(TBB_FOUND TRUE)\n")
file(WRITE "${_prefix}/lib/cmake/ZLIB/ZLIBConfig.cmake"
  "add_library(ZLIB::ZLIB INTERFACE IMPORTED)\nset(ZLIB_FOUND TRUE)\n")
file(WRITE "${_prefix}/lib/cmake/Boost/BoostConfig.cmake"
  "add_library(Boost::iostreams INTERFACE IMPORTED)\nadd_library(Boost::system INTERFACE IMPORTED)\nset(Boost_FOUND TRUE)\n")
file(WRITE "${_prefix}/lib/cmake/IlmBase/IlmBaseConfig.cmake"
  "add_library(IlmBase::Half INTERFACE IMPORTED)\nset(IlmBase_FOUND TRUE)\n")
file(WRITE "${_prefix}/lib/cmake/Imath/ImathConfig.cmake"
  "add_library(Imath::Imath INTERFACE IMPORTED)\nset(Imath_FOUND TRUE)\n")

set(_major 8)
set(_minor 2)
set(_half_define "/* #undef OPENVDB_USE_IMATH_HALF */")
set(_imath_define "/* #undef OPENVDB_IMATH_VERSION */")
set(_expected_half "")
if(FIXTURE_CASE STREQUAL legacy)
  set(_major 7)
  set(_minor 2)
  set(_half_define "")
  set(_expected_half IlmBase::Half)
elseif(FIXTURE_CASE MATCHES "^(ilmbase|missing_ilmbase|defined_zero)$")
  set(_half_define "#define OPENVDB_USE_IMATH_HALF")
  set(_expected_half IlmBase::Half)
  if(FIXTURE_CASE STREQUAL defined_zero)
    # Types.h uses #ifdef, so even a numeric zero selects external Half.
    set(_half_define "#define OPENVDB_USE_IMATH_HALF 0")
  endif()
elseif(FIXTURE_CASE MATCHES "^(imath|missing_imath)$")
  set(_half_define "#define OPENVDB_USE_IMATH_HALF")
  set(_imath_define "#define OPENVDB_IMATH_VERSION 3.1.9")
  set(_expected_half Imath::Imath)
elseif(NOT FIXTURE_CASE STREQUAL embedded AND NOT FIXTURE_CASE STREQUAL installed)
  message(FATAL_ERROR "Unknown fixture case: ${FIXTURE_CASE}")
endif()

if(NOT _expected_half)
  set(CMAKE_DISABLE_FIND_PACKAGE_IlmBase TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_Imath TRUE)
elseif(FIXTURE_CASE STREQUAL missing_ilmbase)
  set(CMAKE_DISABLE_FIND_PACKAGE_IlmBase TRUE)
elseif(FIXTURE_CASE STREQUAL missing_imath)
  set(CMAKE_DISABLE_FIND_PACKAGE_Imath TRUE)
endif()

if(FIXTURE_CASE STREQUAL installed)
  if(NOT EXISTS "${OPENVDB_TEST_PREFIX}/include/openvdb/version.h")
    message(FATAL_ERROR "OPENVDB_TEST_PREFIX must point to an existing installation")
  endif()
  set(OpenVDB_INCLUDE_DIR "${OPENVDB_TEST_PREFIX}/include" CACHE PATH "")
  find_library(_installed_openvdb NAMES openvdb libopenvdb
    PATHS "${OPENVDB_TEST_PREFIX}/lib" "${OPENVDB_TEST_PREFIX}/lib64" NO_DEFAULT_PATH)
  if(NOT _installed_openvdb)
    message(FATAL_ERROR "OPENVDB_TEST_PREFIX does not contain the OpenVDB library")
  endif()
  set(OpenVDB_openvdb_LIBRARY_RELEASE "${_installed_openvdb}" CACHE FILEPATH "")
else()
  file(MAKE_DIRECTORY "${_prefix}/include/openvdb")
  file(WRITE "${_prefix}/include/openvdb/version.h"
    "#define OPENVDB_LIBRARY_MAJOR_VERSION_NUMBER ${_major}\n#define OPENVDB_LIBRARY_MINOR_VERSION_NUMBER ${_minor}\n#define OPENVDB_LIBRARY_PATCH_VERSION_NUMBER 0\n${_half_define}\n${_imath_define}\n")
  file(WRITE "${_prefix}/lib/openvdb.fixture" "Discovery fixture; never linked.\n")
  set(OpenVDB_INCLUDE_DIR "${_prefix}/include" CACHE PATH "")
  set(OpenVDB_openvdb_LIBRARY_RELEASE "${_prefix}/lib/openvdb.fixture" CACHE FILEPATH "")
  set(OpenVDB_ABI "${_major}")
endif()

find_package(OpenVDB 5.0 REQUIRED COMPONENTS openvdb)
get_target_property(_links OpenVDB::openvdb INTERFACE_LINK_LIBRARIES)
get_target_property(_defines OpenVDB::openvdb INTERFACE_COMPILE_DEFINITIONS)
if(_expected_half)
  if(NOT OpenVDB_USES_IMATH_HALF OR NOT _expected_half IN_LIST _links)
    message(FATAL_ERROR "Expected external provider ${_expected_half}; got ${_links}")
  endif()
  if(NOT "OPENVDB_OPENEXR_STATICLIB" IN_LIST _defines)
    message(FATAL_ERROR "Missing legacy external-Half static definition")
  endif()
else()
  if(OpenVDB_USES_IMATH_HALF OR TARGET IlmBase::Half OR TARGET Imath::Imath
      OR "IlmBase::Half" IN_LIST _links OR "Imath::Imath" IN_LIST _links
      OR "OPENVDB_OPENEXR_STATICLIB" IN_LIST _defines)
    message(FATAL_ERROR "Embedded Half unexpectedly requires an external provider")
  endif()
endif()
if(_expected_half STREQUAL "IlmBase::Half" AND TARGET Imath::Imath)
  message(FATAL_ERROR "A legacy-Half build unexpectedly discovered modern Imath")
elseif(_expected_half STREQUAL "Imath::Imath" AND TARGET IlmBase::Half)
  message(FATAL_ERROR "An Imath build unexpectedly discovered legacy IlmBase")
endif()
message(STATUS "Half finder fixture passed: ${FIXTURE_CASE}")
